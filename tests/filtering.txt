# Run with `tern ctl --control DISPOSABLE_ENDPOINT --file tests/filtering.txt` on macOS.
# Requires this plugin in an isolated TERN_CONFIG_DIR under /tmp/tern-filter-*. Never use a live window.
# Local Carly Lua assertions do not invoke a model.
# Native focus and text updates cross the daemon; settle before the next action.
carly lua "local config = tern.getenv('TERN_CONFIG_DIR'); assert(config and config:sub(1,17) == '/tmp/tern-filter-', 'requires disposable filter config'); filter_root = '/tmp/tern-filter-fixtures'; local records = {{name='AlPhA Console',path=filter_root..'/one'},{name='Backend',path=filter_root..'/ALPHA-two'},{name='Regex [a].*',path=filter_root..'/three'},{name='项目😀',path=filter_root..'/four'}}; for _, w in records do tern.fs.mkdir(w.path) end; local file = config..'/plugin-data/workspaces/kv.json'; local data = tern.json.decode(tern.fs.read(file)); data.workspaces = records; tern.fs.write(file,tern.json.encode(data)); for _, pane in cx.session:panes() do if pane.block == 'workspaces.sidebar' then filter_pane = pane.pane end end; assert(filter_pane, 'sidebar must exist'); cx.session:event(filter_pane,{ev='action',act='refresh'}); function check_filter(paths) local sf = cx.session:surface(filter_pane); local found = {}; for _, node in sf.main.c do if node.p and node.p.role == 'workspaces-project' then table.insert(found,node.id) end end; assert(#found == #paths, 'wrong filtered project count'); for _, suffix in paths do assert(table.find(found,'main.workspace:'..filter_root..'/'..suffix),'missing project '..suffix) end; return true end; return true"
carly ok
carly close
